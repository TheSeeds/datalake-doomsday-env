# Dependências de produção do lakecore + camada de catálogo.
#
# `lakecore` em si é stdlib-only (ADR-007). Estas dependências pertencem ao
# **adaptador de catálogo** e ao schema, não ao domínio.

# Schema e migração.
# Versões pinadas: migração que roda com uma versão e falha com outra não é
# migração, é sorte. A subida de versão é evento deliberado, não automático.
SQLAlchemy==2.0.36
alembic==1.14.0

# Driver PostgreSQL **síncrono**, de propósito: o projeto recusou asyncpg antes
# (lição registrada no geoflow) porque não há ganho real neste volume e o custo
# de manter duas pilhas de I/O é alto.
psycopg[binary]==3.2.3

# Ingestão (Onda 4): HTTP + cliente S3.
# `boto3` só é importado por `lakecore.adapters` — o domínio continua stdlib-only.
fastapi==0.115.0
# `starlette`, `uvicorn` e `sse-starlette` pinadas de propósito: `mcp` declara
# pisos frouxos (`starlette>=0.27`, `uvicorn>=0.31.1`), então o resolver sobe
# para as versões mais novas e quebra `fastapi 0.115`, que exige
# `starlette<0.39`. O sintoma é silencioso: o import do FastAPI continua
# funcionando e o ingest-api só falha em runtime. Fixar aqui é o que impede
# "instalou e funcionou ontem" de voltar. O conjunto abaixo foi verificado com
# `pip install --dry-run` como o único compatível com as duas pilhas.
starlette==0.38.6
uvicorn[standard]==0.32.1
sse-starlette==2.1.3
python-multipart==0.0.12
boto3==1.44.5

# Ponte para a tela de chat (Onda 5.3): servidor MCP que expõe o catálogo.
# Só `lakecore.mcp` importa este pacote; o domínio e o executor continuam
# sem saber que existe uma tela.
mcp==1.30.0
duckdb==1.5.6
